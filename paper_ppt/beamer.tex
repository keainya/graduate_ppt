\section{研究背景与意义}

% 第3页 · AI多任务调度的核心矛盾
\begin{frame}{AI多任务调度的核心矛盾}
  \begin{block}{背景}
    人工智能大模型在NLP、CV、智能决策等领域广泛应用，\textbf{多任务调度需求呈爆发式增长}。
  \end{block}

  \begin{alertblock}{矛盾一：AI特性适配性差}
    传统调度系统无法有效处理AI任务的\textbf{异构参数}与\textbf{非结构化数据}，数据治理繁杂。
  \end{alertblock}

  \begin{alertblock}{矛盾二：决策过度依赖人工}
    系统\textbf{无法自动判断}海量AI任务是否具备执行条件，依赖人工逐一启动，浪费人力资源。
  \end{alertblock}

  \begin{alertblock}{矛盾三：缺乏资源互斥检测}
    无法识别GPU等AI临界资源的互斥冲突，任务只能\textbf{低效串行}执行，计算资源严重浪费。
  \end{alertblock}
\end{frame}

% 第4页 · 研究意义
\begin{frame}{研究意义}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{理论意义}
      \begin{itemize}
        \item 填补AI多任务调度领域的\textbf{技术空白}
        \item 提出面向AI特性的调度算法与数据适配机制
        \item 为异构资源互斥检测提供新思路
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{工程价值}
      \begin{itemize}
        \item 已落地\textbf{腾讯TEG}真实业务环境
        \item 月稳定支撑约\textbf{200个}AI多任务运行
        \item 显著提升资源利用率，降低运维人力投入
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.5cm}
  \begin{center}
    \textbf{顺应AI大模型规模化落地的必然需求}
  \end{center}
\end{frame}

    \begin{exampleblock}{核心缺陷}

      \textbf{高算力、长周期、}\\
      \textbf{多模态交互}特性\\

% 第6页 · 任务可用性检测 & 资源互斥检测现状
\begin{frame}{现有检测机制的局限}
  \begin{block}{任务可用性检测 — 逻辑引擎}
    \begin{itemize}
      \item 单调推理 → 非单调推理 → 动态自适应
      \item 均依赖\textbf{静态规则}与预定义知识，缺乏\textbf{综合环境感知}
      \item 无法分析AI任务的目标、资源约束、环境动态性
    \end{itemize}
  \end{block}

  \begin{block}{资源互斥检测 — 传统互斥算法}
    \begin{itemize}
      \item Dekker → Peterson → Lamport时间戳
      \item 面向\textbf{传统同构资源}，无法应对GPU/TPU等\textbf{AI异构资源}
      \item 缺乏对资源需求动态变化、碎片化问题的处理能力
    \end{itemize}
  \end{block}

  \begin{exampleblock}{结论}
    现有技术体系在AI多任务场景下全面失效，亟需全新设计方案。
  \end{exampleblock}
\end{frame}

% ---------- 创新一 ----------
% 第12页 · 创新一（问题+方案）
\begin{frame}{创新一：AI参数画像与数据自动纠错}
  \begin{alertblock}{痛点}
    传统调度系统仅支持结构化数据，无法处理AI任务的\textbf{异构、非结构化参数}，频繁导致任务失败。
  \end{alertblock}

  \begin{exampleblock}{解决方案}
    \textbf{建立大模型任务参数画像 + Pydantic自动纠偏}
    \begin{enumerate}
      \item \textbf{参数画像}：梳理各AI模型参数特征，建立异构数据映射机制
      \item \textbf{自动纠偏逻辑}：
      \begin{itemize}
        \item 全角标点 $\rightarrow$ 半角标点
        \item 冗余符号自动清理
        \item float $\rightarrow$ int 无损类型转换
        \item 参数缺失智能补全
      \end{itemize}
      \item \textbf{JSON/Dict深度映射}：实现多层级参数的精准传递
    \end{enumerate}
  \end{exampleblock}
\end{frame}

\begin{frame}{创新二：基于逻辑引擎的自动化决策}
  \begin{alertblock}{痛点}

  \begin{exampleblock}{解决方案}

      \textbf{条件层（环境感知）}

      \begin{itemize}
        \item 用户自定义调度策略
        \item 配置化管理，无需改代码
        \item 多条件组合判定
      \end{itemize}
    \end{columns}
  \end{exampleblock}

% 第14.5页 · 创新二（核心公式）
\begin{frame}{创新二：逻辑引擎的形式化定义}
  \begin{block}{条件算子（Condition）}
    逻辑判定的最小原子单元，映射关系：
    \[
    C = \{Code, Params\} \rightarrow f_{\text{logic}}(Env, Params)
    \]
    \begin{itemize}
      \footnotesize
      \item $Code$：唯一标识底层判断逻辑（如时间范围、负载阈值）
      \item $Params$：运行时参数，赋予同一逻辑适配不同场景的灵活性
      \item $Env$：实时采集的内外部环境状态
    \end{itemize}
  \end{block}

  \begin{block}{规则算子（Rule）}
    原子条件经逻辑算子嵌套组合而成：
    \[
    Rule = \bigvee_{i=1}^{n} \left( \bigwedge_{j=1}^{m} Condition_{i,j} \right)
    \]
    \begin{itemize}
      \footnotesize
      \item 一级数组元素间执行\textbf{"或（OR）"}逻辑
      \item 二级数组元素间执行\textbf{"且（AND）"}逻辑
      \item 支持"场景策略注入"：节假日、封网期间自动调整准入规则
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{创新三：资源互斥智能分析算法}
  \begin{alertblock}{痛点}
    GPU等AI临界资源存在互斥冲突，传统调度方案只能\textbf{保守串行执行}，资源利用率低。
  \end{alertblock}

  \begin{exampleblock}{解决方案}
    \textbf{细粒度"临界资源表" + 互斥智能检测}
    \begin{enumerate}
      \item \textbf{资源注册}：任务需在定义阶段声明所需临界资源清单
      \item \textbf{状态探测}：实时扫描资源占用表，检测 is\_delete 标识
      \item \textbf{并发决策}：无冲突$\rightarrow$并发执行；有冲突$\rightarrow$延迟重试
    \end{enumerate}
  \end{exampleblock}

  \vspace{0.2cm}
  \begin{center}
    \small
    资源占用表：\textit{task\_id} + \textit{resource\_id} + \textit{is\_delete} $\longrightarrow$ 实时快照
  \end{center}

  \vspace{0.2cm}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{冲突场景}
      同一GPU\#1上多个推理任务$\rightarrow$\textbf{串行排队}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{无冲突场景}
      不同GPU\#1/\#2上独立任务$\rightarrow$\textbf{安全并发}
    \end{block}
  \end{columns}
\end{frame}
